version: '3.8'

services:
  nextjs:
    image: node:18-alpine
    container_name: apex-fit-nextjs-dev
    working_dir: /app
    command: |
      sh -c "
        echo 'Setting environment variables...' &&
        export NEXT_PUBLIC_PYTHON_API_URL=http://localhost:8013 &&
        export NEXT_PUBLIC_API_URL=http://localhost:8013 &&
        echo 'Installing dependencies...' &&
        npm install &&
        echo 'Building Next.js application...' &&
        npm run build &&
        echo 'Starting production server...' &&
        npm start
      "
    ports:
      - "3000:3000"
    environment:
      - NODE_ENV=production
      - PORT=3000
      - HOSTNAME=0.0.0.0
      - NEXT_PUBLIC_PYTHON_API_URL=http://localhost:8013
      - NEXT_PUBLIC_API_URL=http://localhost:8013
    volumes:
      - ./:/app
      - node_modules:/app/node_modules
      - next_cache:/app/.next
    restart: unless-stopped
    depends_on:
      - python-api
      - redis
    networks:
      - apex-fit-network

  python-api:
    image: python:3.11-slim
    container_name: apex-fit-python-api-dev
    working_dir: /app
    command: |
      sh -c "
        apt-get update &&
        apt-get install -y gcc curl &&
        pip install -r requirements.txt &&
        uvicorn main:app --host 0.0.0.0 --port 8000
      "
    ports:
      - "8013:8000"
    environment:
      - PYTHONUNBUFFERED=1
    volumes:
      - ./python-api:/app
    restart: unless-stopped
    depends_on:
      - redis
    networks:
      - apex-fit-network

  redis:
    image: redis:7-alpine
    container_name: apex-fit-redis-dev
    ports:
      - "6380:6379"
    restart: unless-stopped
    networks:
      - apex-fit-network

volumes:
  node_modules:
    driver: local
  next_cache:
    driver: local

networks:
  apex-fit-network:
    driver: bridge
