version: '3.8'

services:
  frontend:
    build:
      context: ./bodyfat-gui-app
      dockerfile: Dockerfile
      args:
        - NODE_ENV=production
        - REACT_APP_API_URL=${API_URL}
    image: bodyfat-gui-app:latest
    container_name: bodyfat-frontend
    ports:
      - "3000:80"
    environment:
      - NODE_ENV=production
      - REACT_APP_API_URL=http://backend:5000
    volumes:
      - ./nginx.conf:/etc/nginx/conf.d/default.conf:ro
      - frontend-build:/usr/share/nginx/html
    depends_on:
      backend:
        condition: service_healthy
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:80/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 40s
    restart: unless-stopped
    networks:
      - bodyfat-network

  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    image: bodyfat-backend:latest
    container_name: bodyfat-backend
    ports:
      - "5000:5000"
    environment:
      - FLASK_ENV=production
      - DATABASE_URL=${DATABASE_URL}
    volumes:
      - ./backend/data:/app/data
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:5000/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 30s
    restart: unless-stopped
    networks:
      - bodyfat-network

volumes:
  frontend-build:
    driver: local

networks:
  bodyfat-network:
    driver: bridge
    ipam:
      config:
        - subnet: 172.28.0.0/16
